Q.1 ----Pet Adoption App---------

CLASS Pet:
    ATTRIBUTE name
    ATTRIBUTE age

    CONSTRUCTOR(name, age):
        SET this.name = name
        SET this.age = age

    METHOD displayInfo():
        PRINT "Name:", name
        PRINT "Age:", age

CLASS Dog EXTENDS Pet:
    ATTRIBUTE breed

    CONSTRUCTOR(name, age, breed):
        CALL super(name, age)
        SET this.breed = breed

    METHOD displayInfo():
        CALL super.displayInfo()
        PRINT "Breed:", breed

CLASS Cat EXTENDS Pet:
    ATTRIBUTE color

    CONSTRUCTOR(name, age, color):
        CALL super(name, age)
        SET this.color = color

    METHOD displayInfo():
        CALL super.displayInfo()
        PRINT "Color:", color

CLASS Bird EXTENDS Pet:
    ATTRIBUTE canFly

    CONSTRUCTOR(name, age, canFly):
        CALL super(name, age)
        SET this.canFly = canFly

    METHOD displayInfo():
        CALL super.displayInfo()
        PRINT "Can Fly:", canFly

MAIN:
    CREATE Dog d1 = NEW Dog("Tommy", 3, "Labrador")
    CREATE Cat c1 = NEW Cat("Kitty", 2, "White")
    CREATE Bird b1 = NEW Bird("Tweety", 1, true)

    CALL d1.displayInfo()
    CALL c1.displayInfo()
    CALL b1.displayInfo()

Q.2.-----Library Membership Mentor----------------

CLASS Member:
    ATTRIBUTE name
    ATTRIBUTE id
    ATTRIBUTE membershipDate
    ATTRIBUTE validityPeriod = 1 year

    CONSTRUCTOR(name, id, membershipDate):
        SET this.name = name
        SET this.id = id
        SET this.membershipDate = membershipDate

    METHOD isMembershipValid(currentDate):
        IF currentDate - membershipDate <= validityPeriod:
            RETURN true
        ELSE:
            RETURN false

    METHOD displayMember():
        PRINT "Name:", name
        PRINT "ID:", id
        PRINT "Membership Date:", membershipDate

MAIN:
    CREATE Member m1 = NEW Member("Alice", "M101", "2024-01-01")
    CALL m1.displayMember()
    CALL m1.isMembershipValid("2025-01-01")
