Q.1----Online Quiz Timer--------

CLASS QuizTimer:
    ATTRIBUTE times[]  // array of question durations
    ATTRIBUTE n        // number of questions
    ATTRIBUTE index    // current question

    CONSTRUCTOR(times[], n):
        this.times = times
        this.n = n
        index = 0

    METHOD nextQuestion():
        PRINT "Question", index+1, "time:", times[index], "seconds"
        IF times[index] <= 5:
            PRINT "⚠️ Warning: Time almost up!"
        index = (index + 1) % n   // circular move

MAIN:
    times = [30, 20, 15]
    CREATE timer = NEW QuizTimer(times, 3)

    FOR i=1 to 6:
        timer.nextQuestion()

Q.2----Job Scheduler---------

CLASS Job:
    ATTRIBUTE name
    ATTRIBUTE deadline
    ATTRIBUTE duration

CLASS JobScheduler:
    ATTRIBUTE priorityQueue   // order by deadline, then duration

    METHOD addJob(job):
        INSERT job INTO priorityQueue

    METHOD runJobs():
        WHILE priorityQueue NOT empty:
            job = REMOVE job WITH highest priority
            PRINT "Running job:", job.name, "Deadline:", job.deadline, "Duration:", job.duration

MAIN:
    CREATE scheduler = NEW JobScheduler()
    scheduler.addJob(Job("Task1", 3, 10))
    scheduler.addJob(Job("Task2", 1, 5))
    scheduler.addJob(Job("Task3", 2, 2))

    scheduler.runJobs()
